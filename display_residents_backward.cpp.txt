#include "hostel_doubly_linked_list.h"
#include <iostream>

void HostelDoublyLinkedList::PrintBackward() const
{
    std::cout << "\nResidents (Last to First):\n";

    for (ResidentNode* node = tail; node != nullptr; node = node->left)
    {
        std::cout << "\nStudent ID: " << node->studentID << '\n';
        std::cout << "Room Number: " << node->roomNumber << '\n';
        std::cout << "Name: " << node->residentName << '\n';
        std::cout << "Department: " << node->department << '\n';
        std::cout << "Phone: " << node->phoneNumber << '\n';
        std::cout << "Guardian: " << node->guardianName << '\n';
        if (node == current)
            std::cout << " [CURRENT]";
        std::cout << "\n-----------------------------\n";
    }

    std::cout << "Total residents: " << Count() << '\n';
}
