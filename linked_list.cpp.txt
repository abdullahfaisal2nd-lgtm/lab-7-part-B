#include "hostel_doubly_linked_list.h"

HostelDoublyLinkedList::HostelDoublyLinkedList()
    : head(nullptr), tail(nullptr), current(nullptr), count(0) {}

HostelDoublyLinkedList::~HostelDoublyLinkedList()
{
    clear();
}

void HostelDoublyLinkedList::clear()
{
    while (head != nullptr)
    {
        current = head;
        delete RemoveCurrent();
    }
}

bool HostelDoublyLinkedList::IsEmpty() const
{
    return head == nullptr;
}

bool HostelDoublyLinkedList::IsNotEmpty() const
{
    return head != nullptr;
}

int HostelDoublyLinkedList::Count() const
{
    return count;
}
