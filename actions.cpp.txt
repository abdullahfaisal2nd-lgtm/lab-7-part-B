#include "actions.h"
#include "hostel_doubly_linked_list.h"
#include "input.h"
#include "menu.h"
#include <iostream>

void runHostelManagementSystem()
{
    HostelDoublyLinkedList residents;
    int choice;

    do
    {
        showHostelMenu();
        std::cin >> choice;

        switch (choice)
        {
        case 1:
            residents.PrintCurrent();
            break;
        case 2:
            residents.Next();
            residents.PrintCurrent();
            break;
        case 3:
            residents.Previous();
            residents.PrintCurrent();
            break;
        case 4:
            residents.InsertRight(readResidentFromUser());
            break;
        case 5:
            residents.InsertAfterCurrent(readResidentFromUser());
            break;
        case 6:
        {
            ResidentNode* removed = residents.RemoveCurrent();
            if (removed != nullptr)
            {
                std::cout << "Removed resident: "
                          << removed->residentName << '\n';
                delete removed;
            }
            else
                std::cout << "Hostel list is empty.\n";
            break;
        }
        case 7:
            residents.PrintForward();
            break;
        case 8:
            residents.PrintBackward();
            break;
        case 9:
            std::cout << "\nHostel Status\n";
            std::cout << "Total residents: " << residents.Count() << '\n';
            std::cout << (residents.IsEmpty() ? "Hostel list is empty.\n"
                                               : "Hostel list is not empty.\n");
            break;
        case 10:
            std::cout << "Exiting Hostel Management System.\n";
            break;
        default:
            std::cout << "Invalid choice.\n";
        }
    } while (choice != 10);
}
