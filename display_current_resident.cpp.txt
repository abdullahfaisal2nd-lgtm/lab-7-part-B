#include "hostel_doubly_linked_list.h"
#include <iostream>

void HostelDoublyLinkedList::PrintCurrent() const
{
    if (current == nullptr)
    {
        std::cout << "Hostel list is empty.\n";
        return;
    }

    std::cout << "\nCurrent Resident Details\n";
    std::cout << "Student ID: " << current->studentID << '\n';
    std::cout << "Room Number: " << current->roomNumber << '\n';
    std::cout << "Name: " << current->residentName << '\n';
    std::cout << "Department: " << current->department << '\n';
    std::cout << "Phone: " << current->phoneNumber << '\n';
    std::cout << "Guardian: " << current->guardianName << '\n';
}
